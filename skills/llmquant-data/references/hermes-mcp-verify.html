Hermes MCP Discovery Check

Run this after restarting Hermes to confirm MCP tool discovery.
Open the browser DevTools Console and paste the snippet. If prefixed tools appear, the server is connected and registered. 

(() => {
    try {
      // Hermes runtime tree isn't guaranteed across versions, so probe the registry shape first.
      const registry = window.__hermes?.toolRegistry ?? window.__hermes__?.toolRegistry;
      if (!registry) return 'No detectable runtime registry object (expected window.__hermes.toolRegistry or window.__hermes__.toolRegistry)';
      const names = Object.keys(registry.tools || registry.entries || registry).sort();
      const mcp = names.filter(n => /^mcp_(llmquant|vscode|obsidian)_/.test(n));
      if (!mcp.length) return 'No MCP tools found.';
      return (
        'Found ' + mcp.length + ' MCP tools:\n' +
        mcp.map(n => '- ' + n).join('\n')
      );
    } catch (e) {
      return 'Error: ' + (e && e.message);
    }
  })()
